```javascript
    function profileToMarkdown(p) {
        const nameL = LANG === 'es' ? 'Nombre' : LANG === 'en' ? 'Name' : 'Nom';
        const langL = LANG === 'es' ? 'Idiomas' : LANG === 'en' ? 'Languages' : 'Langues';
        const styleL = LANG === 'es' ? 'Estilo' : LANG === 'en' ? 'Style' : 'Style';
        const topicL = LANG === 'es' ? 'Temas' : LANG === 'en' ? 'Topics' : 'Sujets';
        const factsL = LANG === 'es' ? 'Datos' : LANG === 'en' ? 'Facts' : 'Infos';
        const goalsL = LANG === 'es' ? 'Objetivos' : LANG === 'en' ? 'Goals' : 'Objectifs';
        const updL = LANG === 'es' ? 'Actualizado' : LANG === 'en' ? 'Updated' : 'Mis à jour';
        const lines = ['# Zova — ' + (LANG === 'es' ? 'Mi perfil' : LANG === 'en' ? 'My profile' : 'Mon profil'), ''];
        if (p.name) lines.push(`**${nameL}:** ${p.name}`, '');
        if (p.languages?.length) lines.push(`**${langL}:** ${p.languages.join(', ')}`, '');
        if (p.style) lines.push(`**${styleL}:** ${p.style}`, '');
        if (p.topics?.length) lines.push(`**${topicL}:** ${p.topics.join(', ')}`, '');
        if (p.facts?.length) lines.push(`## ${factsL}`, ...p.facts.map(f => `- ${f}`), '');
        if (p.goals?.length) lines.push(`## ${goalsL}`, ...p.goals.map(g => `- ${g}`), '');
        if (p.lastUpdated) lines.push('', `_${updL}: ${p.lastUpdated}_`);
        return lines.join('\n');
    }

    async function exportUserProfileMd() {
        const p = getUserProfile();
        if (!profileHasContent(p)) {
            customAlert(t('profile_none'), { title: '👤 ' + t('config_section_profile'), icon: '👤' });
            return;
        }
        await exportJsonFile(profileToMarkdown(p), 'zova-profil.md');
    }
```

```javascript
    function wipeAllAiData() {
        // Exact keys
        ['buddy_memories', 'buddy_userProfile'].forEach(k => localStorage.removeItem(k));
        // Prefix sweep (all personas)
        const prefixes = ['buddy_pmem_', 'buddy_pprofile_', 'buddy_followups_',
            'buddy_bridge_', 'buddy_goals_', 'buddy_chapters_', 'buddy_chapters_ts_'];
        for (let i = localStorage.length - 1; i >= 0; i--) {
            const key = localStorage.key(i);
            if (key && prefixes.some(p => key.startsWith(p))) localStorage.removeItem(key);
        }
    }
```
